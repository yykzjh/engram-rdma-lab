cmake_minimum_required(VERSION 3.18)
project(engram_sge_write LANGUAGES CXX)
find_package(CUDAToolkit REQUIRED)
find_package(MPI REQUIRED COMPONENTS CXX)
find_package(Threads REQUIRED)
find_path(NCCL_INCLUDE_DIR nccl_device.h HINTS ${NCCL_ROOT}/include REQUIRED)
find_library(NCCL_LIBRARY nccl HINTS ${NCCL_ROOT}/lib ${NCCL_ROOT}/lib64 REQUIRED)
set(ENGRAM_GPU_ARCH compute_100 CACHE STRING "PTX target architecture")
file(GLOB_RECURSE NCCL_DEVICE_HEADERS ${NCCL_INCLUDE_DIR}/nccl_device/*.h ${NCCL_INCLUDE_DIR}/nccl_device/*.cuh)
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/gpu_request.ptx
    COMMAND ${CUDAToolkit_NVCC_EXECUTABLE} -std=c++17 -O3 -lineinfo -arch=${ENGRAM_GPU_ARCH} -ptx
    -I${NCCL_INCLUDE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/gpu_request.cu -o ${CMAKE_CURRENT_BINARY_DIR}/gpu_request.ptx
    DEPENDS gpu_request.cu ${NCCL_DEVICE_HEADERS} VERBATIM)
add_custom_target(engram_gpu_ptx DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/gpu_request.ptx)
add_executable(engram_sge_write gpu_request.cpp)
add_dependencies(engram_sge_write engram_gpu_ptx)
add_executable(export_indices export_indices.cpp)
foreach(target engram_sge_write export_indices)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_options(${target} PRIVATE -O3 -Wall -Wextra -Wno-missing-field-initializers)
    target_include_directories(${target} PRIVATE ${NCCL_INCLUDE_DIR} ${CUDAToolkit_INCLUDE_DIRS} ${CUDAToolkit_INCLUDE_DIRS}/cccl)
    target_link_libraries(${target} PRIVATE ibverbs mlx5 CUDA::cuda_driver CUDA::nvrtc Threads::Threads)
endforeach()
target_link_libraries(engram_sge_write PRIVATE MPI::MPI_CXX ${NCCL_LIBRARY})
